 def shortest_path(source, Edges):
 	Distance = { source: 0 }
 	Visited  = { source }
 	Fringe   = Set()
 	Fringe.insert( (0, source) )
 	while not Fringe.isEmpty():
 		d, u = Fringe.pop() # get and remove smallest element
 		for v, l in Edges[u]:
 			dv = Distance.get(v, None)
 			if dv == None or d + l < dv:
 				if dv != None:
 					Fringe.delete( (dv, v) )
 				Distance[v] = d + l
 				Fringe.insert( (d + l, v) )
 		Visited.add(u)
 	return Distance
\end{minted}

Der TypeScript Code:
Bitte den Code zuerst erklären
\Needspace{31\baselineskip}
\begin{minted}{typescript}
	function shortestPath(
	source: string,
	Edges: Record<string, Array<[string, number]>>
	): Record<string, number> {
		const Distance: Record<string, number> = { [source]: 0 };
		const visited: Record<string, boolean> = {};
		const Fringe = new AVLSet<[number, string]>();
		Fringe.insert([0, source]);
		while (!Fringe.isEmpty()) {
			const [d, u] = Fringe.pop();
			if (visited[u]) {
				continue; 
			}
			visited[u] = true;
			
			for (const [v, l] of Edges[u]) {
				const dv = Distance[v];
				if (dv === undefined || d + l < dv) {
					if (dv !== undefined) {
						Fringe.delete([dv, v]);
					}
					Distance[v] = d + l;
					Fringe.insert([d + l, v]);
				}
			}
		}
		return Distance;
	}
\end{minted}
